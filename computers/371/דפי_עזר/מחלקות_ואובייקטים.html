כאשר שם הפרמטר זהה לשם השדה, this.grade מתייחס לשדה של האובייקט, ו־grade מתייחס לפרמטר.

{
    int sum = 0;
    int count = 0;

    for (int i = 0; i < arr.Length; i++)
    {
        if (arr[i] != null)
        {

10. הוספה והזזה לפי count

כאשר מנהלים מערך צפוף, כלומר כל האובייקטים נמצאים מתחילת המערך ברצף, נשתמש במשתנה count. הערכים החוקיים נמצאים בטווח 0 ≤ i < count.

הוספה לסוף האזור הפעיל

static bool AddWithCount(Student[] arr, ref int count, Student s)
static bool AddWithCount(Student[] arr, ref int count, Student s)
{
    if (count == arr.Length)
        return false;

    arr[count] = s;
    count++;
    return true;
}

הכנסה למקום מסוים עם הזזה ימינה

static bool InsertAt(Student[] arr, ref int count, int index, Student s)
static bool InsertAt(Student[] arr, ref int count, int index, Student s)
{
    if (count == arr.Length || index < 0 || index > count)
        return false;

    for (int i = count; i > index; i--)
    {
        arr[i] = arr[i - 1];
    }

    arr[index] = s;
    count++;
    return true;
}

12. העתקת מערך אובייקטים: העתקה רדודה מול עמוקה

העתקה רדודה — מעתיקה הפניות

כל תא במערך החדש מצביע לאותו אובייקט כמו במערך המקורי.

static Student[] ShallowCopy(Student[] arr)
{
    Student[] copy = new Student[arr.Length];

    for (int i = 0; i < arr.Length; i++)
    {
        copy[i] = arr[i];
    }

    return copy;
}

העתקה עמוקה — יוצרת אובייקטים חדשים

משתמשים בפעולה הבונה המעתיקה של Student.

static Student[] DeepCopy(Student[] arr)
{
    Student[] copy = new Student[arr.Length];

    for (int i = 0; i < arr.Length; i++)
    {
        if (arr[i] != null)
            copy[i] = new Student(arr[i]);
    }

    return copy;
}

13. הכלת מערך אובייקטים בתוך מחלקה

מחלקת Course שמכילה מערך תלמידים

זוהי דוגמה למחלקה שמחזיקה בתוכה מערך אובייקטים ומנהלת אותו בעזרת פעולות פנימיות.

public class Course
{
    private string courseName;
    private Student[] students;
    private int count;

    public Course(string courseName, int maxStudents)
    {
        this.courseName = courseName;
        this.students = new Student[maxStudents];
        this.count = 0;
    }

    public Course(Course other)
    {
        this.courseName = other.courseName;
        this.students = new Student[other.students.Length];
        this.count = other.count;

        for (int i = 0; i < other.students.Length; i++)
        {
            if (other.students[i] != null)
                this.students[i] = new Student(other.students[i]);
        }
    }

    public bool Add(Student s)
    {
        if (count == students.Length)
            return false;

        students[count] = new Student(s);
        count++;
        return true;
    }

    public int GetCount()
    {
        return count;
    }
}

פעולה פנימית שסורקת את המערך שבתוך המחלקה

public int CountPassing()
{
    int total = 0;

    for (int i = 0; i < count; i++)
    {
        if (students[i].IsPassing())
            total++;
    }

    return total;
}

{

        return null;

    return new Student(students[index]);
}

15. תבניות מניפולציה נפוצות — סיכום מהיר

משימהרעיון אלגוריתמידגש
סריקה והדפסהלולאה אחת על כל התאים.אם אין count, לבדוק null.
ספירה לפי תנאימונה count שמתקדם רק אם התנאי מתקיים.לדוגמה: עוברים, מצטיינים, שם מסוים.
סכום / ממוצעצובר sum ומונה אובייקטים תקינים.לא לחלק ב־0 אם אין אובייקטים.
חיפושאינדקס מתחיל ב־-1 ונעצור כשהוא נמצא.אפשר להשתמש בתנאי index == -1 בלולאה.
מקסימום / מינימוםשומרים הפניה לאובייקט הטוב ביותר עד עכשיו.מתחילים ב־null.
הוספהלמקום פנוי ראשון או לסוף האזור הפעיל.אם עובדים עם count, לבדוק שלא עברנו את הגודל.
מחיקהאו הצבת null, או הזזה שמאלה והקטנת count.במערך צפוף עדיף הזזה.
מיוןשתי לולאות והשוואה לפי שדה.לא להשוות תאים שהם null.
כפילויותשתי לולאות: j = i + 1.בודקים כל זוג פעם אחת בלבד.
העתקההעתקה רדודה או עמוקה.למערך אובייקטים בדרך כלל עדיף deep copy.
