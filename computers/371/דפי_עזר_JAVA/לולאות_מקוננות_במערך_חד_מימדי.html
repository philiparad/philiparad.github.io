1. מפת התבניות במערך חד־ממדי

מצבתבנית מתאימהמתי משתמשים?
כל זוג איברים פעם אחתfor i → 0..n-1, for j → i+1..n-1כפילויות, זוג שסכומו נתון, הפרש בין שני איברים, השוואת כל זוג בלי חזרה.

השוואה לאיברים שאחריוfor j → i+1..n-1בדיקה קדימה, מציאת כפילות עתידית, זוגות ללא חזרה.

השוואת שני מערכיםfor i over a, for j over bחיתוך, הכלה, האם איבר ממערך אחד מופיע במערך אחר.

2. תבנית כל הזוגות במערך

כל זוג איברים פעם אחת — בלי חזרות

הפעולה מדפיסה כל זוג אינדקסים פעם אחת בלבד. j תלויה ב־i ומתחילה מ־i + 1.

static void printAllPairs(int[] arr)
{
    for (int i = 0; i < arr.length; i++)
    {
        for (int j = i + 1; j < arr.length; j++)
        {
            System.out.println(arr[i] + " , " + arr[j]);
        }
    }
}
דוגמה: עבור {4, 7, 2} יודפסו הזוגות: 4,7, 4,2, 7,2.

האם קיימת כפילות במערך?

static boolean hasDuplicate(int[] arr)
{
    boolean found = false;

    for (int i = 0; i < arr.length; i++)
    {
        for (int j = i + 1; j < arr.length; j++)
        {

}

4. זוגות עם תנאי

האם יש זוג שסכומו שווה ליעד?

בודקים כל זוג פעם אחת. אין צורך לבדוק את אותו זוג בסדר הפוך.

static boolean hasPairWithSum(int[] arr, int target)
{
    boolean found = false;

    for (int i = 0; i < arr.length; i++)
    {
        for (int j = i + 1; j < arr.length; j++)
        {
            if (arr[i] + arr[j] == target)
                found = true;
        }
    }
    return found;
}

static int countPairsWithEvenSum(int[] arr)
{
    int count = 0;

    for (int i = 0; i < arr.length; i++)
    {
        for (int j = i + 1; j < arr.length; j++)
        {

    return count;

הפרש מקסימלי בין שני איברים

בודקים כל זוג ומעדכנים מקסימום. מתאים כאשר רוצים להשוות יחסים בין כל זוג איברים.

static int maxPairDifference(int[] arr)
{
    int maxDiff = 0;

    for (int i = 0; i < arr.length; i++)
    {
        for (int j = i + 1; j < arr.length; j++)
        {
            int diff = Math.abs(arr[i] - arr[j]);
            if (diff > maxDiff)
                maxDiff = diff;
        }
    }
    return maxDiff;
}

static int mostFrequentValue(int[] arr)
{
    int bestValue = arr[0];
    int bestCount = 0;

    for (int i = 0; i < arr.length; i++)
    {
        int count = 0;

        for (int j = 0; j < arr.length; j++)
        {
            if (arr[j] == arr[i])
                count++;
        }

        if (count > bestCount)
        {
            bestCount = count;
            bestValue = arr[i];
        }
    }
    return bestValue;
}

static void printAllSubArrays(int[] arr)
{
    for (int start = 0; start < arr.length; start++)
    {
        for (int end = start; end < arr.length; end++)
        {
            System.out.print("[");
            for (int k = start; k <= end; k++)

                System.out.print(arr[k]);
                if (k < end)

    }
}

static int maxSubArraySumNested(int[] arr)
{

    for (int start = 0; start < arr.length; start++)

}

ממוצע מקסימלי של חלון באורך k

מחשבים סכום לכל חלון, ואז משווים למקסימום.

static double maxWindowAverage(int[] arr, int k)
{
    int bestSum = 0;
    boolean first = true;

    for (int start = 0; start <= arr.length - k; start++)
    {
        int sum = 0;
        for (int j = start; j < start + k; j++)
            sum += arr[j];

        if (first || sum > bestSum)
        {
            bestSum = sum;
            first = false;
        }
    }
    return (double)bestSum / k;
}
הנחה: 1 ≤ k ≤ arr.length.

9. מיון בעזרת לולאות מקוננות

Selection Sort — בחירת הקטן בכל סיבוב

בכל סיבוב הלולאה החיצונית קובעת את המקום הבא למילוי. הפנימית מחפשת את האיבר הקטן ביותר מימין.

static void selectionSort(int[] arr)
{
    for (int i = 0; i < arr.length - 1; i++)
    {
        int minIndex = i;

        for (int j = i + 1; j < arr.length; j++)
        {
            if (arr[j] < arr[minIndex])
                minIndex = j;
        }

        int temp = arr[i];
        arr[i] = arr[minIndex];
        arr[minIndex] = temp;
    }
}
תבנית: i = המקום שמסדרים, j = חיפוש בתחום הלא־מסודר.

Bubble Sort — החלפת שכנים

בכל מעבר מעבירים ערך גדול ימינה באמצעות השוואת שכנים.

static void bubbleSort(int[] arr)
{
    for (int i = 0; i < arr.length - 1; i++)
    {
        for (int j = 0; j < arr.length - 1 - i; j++)
        {
            if (arr[j] > arr[j + 1])
            {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}
קשר בין הלולאות: בכל סיבוב חיצוני התחום של j מתקצר, כי הסוף כבר מסודר.

בדיקה אם המערך ממוין באמצעות כל זוג

בודקים שכל איבר מוקדם אינו גדול מאיבר מאוחר. זו תבנית זוגות, לא הדרך הקצרה ביותר, אך היא מדגימה קשר בין כל זוגות האינדקסים.

static boolean isSortedByPairs(int[] arr)
{
    boolean sorted = true;

    for (int i = 0; i < arr.length; i++)
    {
        for (int j = i + 1; j < arr.length; j++)
        {
            if (arr[i] > arr[j])
                sorted = false;
        }
    }
    return sorted;
}
הבנה: אם קיים זוג שבו מוקדם גדול ממאוחר, המערך אינו ממוין.

האם כל איברי a קיימים בתוך b?

לכל איבר ב־a מחפשים הופעה ב־b. אם איבר אחד לא נמצא — התוצאה הכללית שקרית.

static boolean allValuesOfAInB(int[] a, int[] b)
{
    boolean allFound = true;

    for (int i = 0; i < a.length; i++)
    {
        boolean foundCurrent = false;

        for (int j = 0; j < b.length; j++)
        {
            if (a[i] == b[j])
                foundCurrent = true;
        }

        if (!foundCurrent)
            allFound = false;
    }
    return allFound;
}
תבנית: אמת כללית + בדיקת אמת מקומית לכל איבר.

11. שכנים וסביבה במערך חד־ממדי

סכום שכנים בטווח k סביב כל איבר

לכל מקום i עוברים על תחום קטן סביבו. הגבולות תלויים ב־i.

static void printNeighborSums(int[] arr, int k)
{
    for (int i = 0; i < arr.length; i++)
    {
        int sum = 0;
        int from = i - k;
        int to = i + k;

        if (from < 0)
            from = 0;
        if (to >= arr.length)
            to = arr.length - 1;

        for (int j = from; j <= to; j++)
        {
            if (j != i)
                sum += arr[j];
        }

        System.out.println("index " + i + ": " + sum);
    }
}
שימוש: שאלות על שכנים, מרחקים, וסביבה מקומית במערך.

האם כל איבר גדול מכל הקודמים לו?

לכל i בודקים את כל המקומות שלפניו. אם אחד מהם גדול או שווה — התנאי נכשל.

static boolean eachValueGreaterThanAllPrevious(int[] arr)
{
    boolean ok = true;

    for (int i = 1; i < arr.length; i++)
    {
        for (int j = 0; j < i; j++)
        {
            if (arr[i] <= arr[j])
                ok = false;
        }
    }
    return ok;
}

12. לולאה פנימית לפי ערך האיבר

הדפסת עמודות כוכבים לפי ערכי המערך

כאן מספר הריצות של הלולאה הפנימית נקבע לפי arr[i], ולא לפי אורך המערך.

static void printBars(int[] arr)
{
    for (int i = 0; i < arr.length; i++)
    {
        System.out.print(i + ": ");

        for (int j = 0; j < arr[i]; j++)
        {
            System.out.print("*");
        }

        System.out.println();
    }
}
דוגמה: עבור {3, 1, 4} יודפסו 3 כוכבים, אחר כך 1, אחר כך 4.

סכום חוזר לפי ערכי המערך

לפעמים האיבר עצמו מגדיר כמה פעמים לבצע פעולה.

static int sumRepeatedIndexes(int[] arr)
{
    int total = 0;

    for (int i = 0; i < arr.length; i++)
    {
        for (int j = 0; j < arr[i]; j++)
        {
            total += i;
        }
    }
    return total;
}
תבנית: j < arr[i] — תלות בערך ולא רק באינדקס.

13. שלשות איברים

האם יש שלשה שסכומה שווה ליעד?

כאשר צריך לבדוק שלושה איברים שונים, משתמשים בשלוש לולאות. כל לולאה מתחילה אחרי הקודמת כדי למנוע חזרות.

static boolean hasTripletWithSum(int[] arr, int target)
{
    boolean found = false;

    for (int i = 0; i < arr.length; i++)
    {
        for (int j = i + 1; j < arr.length; j++)
        {
            for (int k = j + 1; k < arr.length; k++)
            {
                if (arr[i] + arr[j] + arr[k] == target)
                    found = true;
            }
        }
    }
    return found;
}
כלל: i < j < k מבטיח שכל שלשה תיבדק פעם אחת בלבד.

static int sumPairsWhereFirstSmaller(int[] arr)
{
    int sum = 0;

    for (int i = 0; i < arr.length; i++)
    {
        for (int j = i + 1; j < arr.length; j++)
        {
            if (arr[i] < arr[j])
                sum += arr[i] + arr[j];
        }
    }
    return sum;
}
