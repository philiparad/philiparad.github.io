public static void fillRnd(int[] arr, int min, int max)

מערך ב־Java אינו גדל במקום. יוצרים מערך חדש ומעתיקים אליו את הערכים האפשריים.
public static int[] resize(int[] arr, int size)
public static int[] resize(int[] arr, int size)
{
    int[] newArr = new int[size];
    int len = Math.min(arr.length, newArr.length);

    for (int i = 0; i < len; i++)
    {
        newArr[i] = arr[i];
    }
    return newArr;
}

public static int[] commonValues(int[] arr1, int[] arr2)
public static int[] commonValues(int[] arr1, int[] arr2)
{
    int[] result = new int[Math.min(arr1.length, arr2.length)];
    int count = 0;

    for (int i = 0; i < arr1.length; i++)
    {
        if (!isExist(result, arr1[i]) && isExist(arr2, arr1[i]))
        {
            result[count] = arr1[i];
            count++;
        }
    }
    return resize(result, count);
}

public static int maxSequence(int[] arr)
{
    if (arr.length == 0)
    {
        return 0;

    for (int i = 0; i < arr.length - 1; i++)

כל רצף נשמר כזוג: ערך, כמות.
public static int[] zip(int[] arr)
public static int[] zip(int[] arr)
{
    if (arr.length == 0)
    {
        return new int[0];
    }

    int[] tmp = new int[arr.length * 2];
    int p = 0;
    int len = 1;

    for (int i = 0; i < arr.length - 1; i++)
    {
        if (arr[i] == arr[i + 1])
        {
            len++;
        }
        else
        {
            tmp[p] = arr[i];
            tmp[p + 1] = len;
            p = p + 2;
            len = 1;
        }
    }

    tmp[p] = arr[arr.length - 1];
    tmp[p + 1] = len;
    p = p + 2;

    return resize(tmp, p);
}
int[] a = { 4, 4, 4, 7, 7, 2 }; // zip(a) -> { 4, 3, 7, 2, 2, 1 }

public static int[] unzip(int[] arr)
public static int[] unzip(int[] arr)
{
    int[] tmp = new int[sumOdd(arr)];
    int j = 0;

    for (int i = 0; i < arr.length; i = i + 2)
    {
        int value = arr[i];
        int amount = arr[i + 1];

        for (int n = 0; n < amount; n++)
        {
            tmp[j + n] = value;
        }
        j = j + amount;
    }
    return tmp;
}

selectionSort - מיון בחירה בסדר יורד

בכל סיבוב מוצאים את הערך הגדול ביותר בחלק שעדיין לא סודר ומחליפים אותו עם המקום הנוכחי.

public static void selectionSort(int[] arr)
{
    for (int i = 0; i < arr.length - 1; i++)
    {
        int imax = iMax(arr, i);
        swap(arr, i, imax);
    }
}

12.מיון בועות

bubbleSort - מיון בועות בסדר עולה

בכל מעבר בודקים זוגות סמוכים. אם זוג אינו מסודר, מחליפים אותו. בסוף כל מעבר, הערך הגדול ביותר בחלק הנבדק מגיע לסופו.
public static void bubbleSort(int[] arr)
public static void bubbleSort(int[] arr)
{
    for (int end = arr.length - 1; end > 0; end--)
    {
        bubblePass(arr, end);
    }
}

private static void bubblePass(int[] arr, int end)
{
    for (int i = 0; i < end; i++)
    {
        if (arr[i] > arr[i + 1])
        {
            swap(arr, i, i + 1);
        }
    }
}
int[] a = { 4, 9, 1, 7 }; bubbleSort(a); // { 1, 4, 7, 9 }
