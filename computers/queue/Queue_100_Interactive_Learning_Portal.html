{

public static int DistanceOccurrences(Queue<double> q, double x)
{
    Queue<double> temp = new Queue<double>();
    int index = 0;
    int first = -1;
    int last = -1;
    while (!q.IsEmpty())
    {
        double value = q.Remove();
        temp.Insert(value);
        if (value == x)
        {
            if (first == -1)
                first = index;
            last = index;
        }
        index++;
    }
    while (!temp.IsEmpty())
        q.Insert(temp.Remove());
    if (first == -1)
        return -1;
    return last - first;
}

                                            Queue<int> b)
{

    while (!a.IsEmpty() && !b.IsEmpty())

        if (a.Head() < b.Head())
            result.Insert(a.Remove());
        else if (b.Head() < a.Head())

    {

        temp.Insert(value);
        if (!hasPrevious)

            previous = value;
            count = 1;
            hasPrevious = true;

            if (count >= 2) result.Insert(previous * count);

            count = 1;

    return result;

{
    if (k <= 0)
        return false;
    Queue<int> temp = new Queue<int>();
    int n = 0;
    while (!q.IsEmpty())
    {
        temp.Insert(q.Remove());
        n++;
    }
    int index = 0;
    long firstSum = 0;
    long lastSum = 0;
    while (!temp.IsEmpty())
    {
        int x = temp.Remove();
        if (index < k)
            firstSum += x;
        if (index >= n - k)
            lastSum += x;
        q.Insert(x);
        index++;
    }
    return n >= 2 * k && firstSum == lastSum;
}

        copy.Insert(value);
        temp.Insert(value);

    while (!temp.IsEmpty())
        q.Insert(temp.Remove());
    if (copy.IsEmpty()) return false;

    {

        bool hasValue = false;
        while (!copy.IsEmpty() && sum < target)

public static Node<int> BuildNumbersList(
    Queue<int> q, Node<int> positions)
{
    Node<int> first = null;
    Node<int> last = null;
    Node<int> current = positions;
    int number = 0;

    {
        int p = current.GetValue();
        if (p == 0)
        {
            Node<int> node = new Node<int>(number);
            if (first == null)
                first = node;
            else

            number = 0;

        first = finalNode;

    return first;
}

public static Queue<RationNumber> CombineDenominators(
    Queue<RationNumber> q)
{
    Queue<RationNumber> temp = new Queue<RationNumber>();
    Queue<RationNumber> work = new Queue<RationNumber>();
    Queue<RationNumber> result = new Queue<RationNumber>();
    while (!q.IsEmpty())
    {
        RationNumber x = q.Remove();
        temp.Insert(x);
        work.Insert(x);
    }
    while (!temp.IsEmpty()) q.Insert(temp.Remove());
    while (!work.IsEmpty())
    {
        RationNumber first = work.Remove();
        int denominator = first.GetDenominator();
        int numerator = first.GetNumerator();
        Queue<RationNumber> remaining = new Queue<RationNumber>();
        while (!work.IsEmpty())

            if (x.GetDenominator() == denominator)

    }

}

{
    int n = Size(q);
    if (m <= 1 || m >= n) return false;
    Queue<int> temp = new Queue<int>();
    int prev = 0, current = 0;
    bool isMagic = false;
    int position = 1;
    while (!q.IsEmpty())
    {

        temp.Insert(x);
        if (position == m - 1) prev = x;

{

    Queue<int> copy = new Queue<int>();
    while (!q.IsEmpty()) temp.Insert(q.Remove());
    while (!temp.IsEmpty())
    {
        int value = temp.Remove();
        q.Insert(value);
        copy.Insert(value);
    }
    return copy;
}

public static bool IsGreaterThanAllFollowing(Queue<int> q, int x)
{
    Queue<int> temp = new Queue<int>();
    bool greater = true;
    while (!q.IsEmpty())
    {
        int value = q.Remove();
        if (value >= x) greater = false;
        temp.Insert(value);
    }
    while (!temp.IsEmpty()) q.Insert(temp.Remove());
    return greater;
}

public static int CountEvaporationRounds(Queue<int> q)

public static Queue<Cloth> MakePairs(Queue<Cloth> q)
{
    Queue<Cloth> waiting = new Queue<Cloth>();
    Queue<Cloth> pairs = new Queue<Cloth>();
    Queue<Cloth> temp = new Queue<Cloth>();
    while (!q.IsEmpty())
    {
        Cloth item = q.Remove();
        Cloth match = null;
        while (!waiting.IsEmpty())
        {
            Cloth candidate = waiting.Remove();
            if (match == null &&
                candidate.GetColor() == item.GetColor() &&
                candidate.GetClothType() != item.GetClothType())
                match = candidate;
            else
                temp.Insert(candidate);
        }
        while (!temp.IsEmpty())
            waiting.Insert(temp.Remove());
        if (match == null)
            waiting.Insert(item);
        else if (item.GetClothType() == "shirt")
        {
            pairs.Insert(item);
            pairs.Insert(match);
        }
        else
        {
            pairs.Insert(match);
            pairs.Insert(item);
        }
    }
    return pairs;
}

            return fastQueue.Remove();

public static int PagesBefore(Queue<PrintingWork> q, PrintingWork work)
{
    Queue<PrintingWork> temp = new Queue<PrintingWork>();
    int pages = 0;
    while (!q.IsEmpty())
    {
        PrintingWork current = q.Remove();
        if (work.GetCopyType() == "clr" ||
            current.GetCopyType() == "bw")
            pages += current.GetPages();
        temp.Insert(current);
    }
    while (!temp.IsEmpty())
        q.Insert(temp.Remove());
    return pages;
}

public static int AssignToPrinter(
    Queue<PrintingWork>[] printers, PrintingWork work)
{
    int best = 0;
    int minimum = int.MaxValue;
    int i = 0;
    while (i < printers.Length)
    {
        int pages = PagesBefore(printers[i], work);
        if (pages <= minimum)
        {
            minimum = pages;
            best = i;
        }
        i++;
    }

    Queue<PrintingWork> selected = printers[best];
    Queue<PrintingWork> temp = new Queue<PrintingWork>();
    bool inserted = false;
    while (!selected.IsEmpty())
    {
        PrintingWork current = selected.Remove();
        if (!inserted && work.GetCopyType() == "bw" &&
            current.GetCopyType() == "clr")
        {
            temp.Insert(work);
            inserted = true;
        }
        temp.Insert(current);
    }
    if (!inserted)
        temp.Insert(work);
    while (!temp.IsEmpty())
        selected.Insert(temp.Remove());
    return best;
}

{
    Queue<Patient> temp = new Queue<Patient>();

    while (!q.IsEmpty())

            inserted = true;
