כלל מרכזי: בפירוק מספר לספרות משתמשים ב־num % 10 כדי לקבל את ספרת האחדות, וב־num / 10 כדי למחוק את ספרת האחדות. לדוגמה: 472 % 10 = 2, 472 / 10 = 47.

תבנית מעבר משמאל לימין

מתאימה כאשר חייבים לשמור את סדר הספרות המקורי, למשל מחיקת ספרה או החלפת ספרה.

int n = num;
if (n < 0)
    n = -n;

int power = 1;
while (power <= n / 10)
    power = power * 10;

while (power > 0)
{
    int digit = n / power;  // הספרה השמאלית הנוכחית

    // טיפול בספרה digit

    n = n % power;          // מחיקת הספרה משמאל
    power = power / 10;
}

// digit חייב להיות בין 0 ל־9

static int countDigit(int num, int digit)
{

        return 0;

    int n = absNumber(num);

    if (n == 0)
    {
        if (digit == 0)

static int reverseNumber(int num)
{
    int n = absNumber(num);
    int result = 0;

    while (n > 0)
    {
        int digit = n % 10;
        result = result * 10 + digit;
        n = n / 10;

    return result;
}

מחיקת כל ההופעות של ספרה

כדי לשמור את הסדר המקורי עוברים משמאל לימין.

static int removeDigit(int num, int digitToRemove)
static int removeDigit(int num, int digitToRemove)
{
    if (!isDigitValue(digitToRemove))
        return num;

    int n = absNumber(num);
    int result = 0;

    int power = 1;
    while (power <= n / 10)
        power = power * 10;

    while (power > 0)
    {
        int digit = n / power;
        if (digit != digitToRemove)
            result = result * 10 + digit;

        n = n % power;
        power = power / 10;
    }

    if (num < 0)
        return -result;

    return result;
}
דוגמה: System.out.println(removeDigit(40720, 0)); // 472

החלפת ספרה בספרה אחרת

static int replaceDigit(int num, int oldDigit, int newDigit)
static int replaceDigit(int num, int oldDigit, int newDigit)
{
    if (!isDigitValue(oldDigit) || !isDigitValue(newDigit))
        return num;

    int n = absNumber(num);
    int result = 0;

    int power = 1;
    while (power <= n / 10)
        power = power * 10;

    while (power > 0)
    {
        int digit = n / power;
        if (digit == oldDigit)
            digit = newDigit;

        result = result * 10 + digit;
        n = n % power;
        power = power / 10;
    }

    if (num < 0)
        return -result;

    return result;
}

static boolean digitsIncreasing(int num)
static boolean digitsIncreasing(int num)
{
    int n = absNumber(num);

    if (n < 10)
        return true;

    int power = 1;
    while (power <= n / 10)
        power = power * 10;

    int prev = n / power;
    n = n % power;
    power = power / 10;

    boolean ok = true;
    while (power > 0)
    {
        int digit = n / power;
        if (digit <= prev)
            ok = false;

        prev = digit;
        n = n % power;
        power = power / 10;
    }
    return ok;
}

7. פירוק לגורמים ולמחלקים

לפעמים “מרכיבי המספר” הם לא הספרות אלא המחלקים או הגורמים הראשוניים שלו. כאן משתמשים בחילוק ושארית, לא בפירוק ספרות.

בדיקה אם מספר ראשוני

static boolean isPrime(int num)
static boolean isPrime(int num)
{
    if (num < 2)
        return false;

    boolean prime = true;
    for (int d = 2; d * d <= num; d++)
    {
        if (num % d == 0)
            prime = false;
    }
    return prime;
}
דוגמה: System.out.println(isPrime(29)); // true

ספירת מחלקים

static int countDivisors(int num)
static int countDivisors(int num)
{
    int n = absNumber(num);

    if (n == 0)
        return 0;

    int count = 0;
    for (int d = 1; d <= n; d++)
    {
        if (n % d == 0)
            count++;
    }
    return count;
}
דוגמה: System.out.println(countDivisors(12)); // 6

האם כל הגורמים הראשוניים שונים?

בכל פעם שמוצאים גורם, סופרים כמה פעמים הוא מחלק את המספר.

static boolean hasNoRepeatedPrimeFactor(int num)
static boolean hasNoRepeatedPrimeFactor(int num)
{
    int n = absNumber(num);
    boolean ok = true;

    for (int d = 2; d * d <= n; d++)
    {
        int count = 0;
        while (n % d == 0)
        {
            count++;
            n = n / d;
        }

        if (count > 1)
            ok = false;
    }
    return ok;
}
דוגמה: System.out.println(hasNoRepeatedPrimeFactor(30)); // true

static int sumPrimeFactors(int num)
static int sumPrimeFactors(int num)
{
    int n = absNumber(num);
    int sum = 0;

    for (int d = 2; d * d <= n; d++)
    {
        while (n % d == 0)
        {
            sum = sum + d;
            n = n / d;
        }
    }

    if (n > 1)
        sum = sum + n;

    return sum;
}
