1. מפת התבניות

מצבתבנית מתאימהמתי משתמשים?
כל התאים במטריצהfor i → rows, for j → colsהדפסה, קלט, סכום, חיפוש, ספירה, שינוי ערכים.
כל זוג איברים במערךfor i → n, for j → i + 1בדיקת כפילויות, זוג שסכומו נתון, השוואת כל זוג פעם אחת.
טווחים רצופיםfor start → n, for end → startסכומי תתי־מערכים, בדיקת רצפים, תחומי התחלה וסוף.
משולש במטריצהj < i / j <= i / j > iאלכסון, חצי תחתון, חצי עליון, הימנעות מחזרה כפולה.
צוברים לפי שורהrowSum = 0 inside outer loopסכום שורה, ממוצע שורה, מקסימום שורה.
צובר כלליtotal = 0 before all loopsסכום כל המטריצה או ספירה כללית.
השוואת שני מערכיםfor i over a, for j over bחיתוך, הכלה, האם איבר מופיע במערך אחר.
שכנים בגרידfor dr = -1..1, for dc = -1..1ספירת שכנים, לוחות משחק, מטריצות סביב תא.

כללי עבודה קבועים

1תמיד לתת שמות שונים למונים: i לשורות או לאיבר ראשון, j לעמודות או לאיבר שני.
2לאתחל צובר פנימי בתוך הלולאה החיצונית כאשר רוצים סכום לכל שורה או לכל איבר.
3כאשר בודקים זוגות, בדרך־כלל j = i + 1, כדי לא להשוות איבר עם עצמו ולא לבדוק אותו זוג פעמיים.
4במטריצה ריבועית בלבד אפשר לדבר על אלכסונים ראשיים ומשניים.

3. צוברים וספירות במטריצות

סכום כל התאים

צובר כללי מאותחל לפני שתי הלולאות, כי הוא שייך לכל המטריצה.

static int SumMatrix(int[][] mat)
{
    int sum = 0;

    for (int i = 0; i < mat.length; i++)
    {
        for (int j = 0; j < mat[0].length; j++)
        {
            sum += mat[i][j];
        }
    }
    return sum;
}
דוגמה: אם המטריצה היא {{1,2},{3,4}}, הסכום הוא 10.

סכום כל שורה

4. חיפוש, מקסימום ומיקום במטריצה

בדיקה האם ערך קיים במטריצה

עוברים על כל התאים. משתמשים בדגל בוליאני כדי לזכור אם הערך נמצא.

static boolean ContainsValue(int[][] mat, int value)
{
    boolean found = false;

    for (int i = 0; i < mat.length; i++)
    {
        for (int j = 0; j < mat[0].length; j++)
        {
            if (mat[i][j] == value)
                found = true;
        }
    }
    return found;
}

שורה עם הסכום הגדול ביותר

לולאה פנימית מחשבת סכום של שורה אחת; אחרי שהיא מסתיימת משווים לסכום הטוב ביותר.

static int RowWithMaxSum(int[][] mat)
{
    int bestRow = 0;
    int bestSum = 0;

    for (int j = 0; j < mat[0].length; j++)
        bestSum += mat[0][j];

    for (int i = 1; i < mat.length; i++)
    {
        int rowSum = 0;

        for (int j = 0; j < mat[0].length; j++)
            rowSum += mat[i][j];

        if (rowSum > bestSum)
        {
            bestSum = rowSum;
            bestRow = i;
        }
    }
    return bestRow;
}
תבנית חשובה: חישוב פנימי לכל שורה + החלטה חיצונית אחרי סיום השורה.

אלכסון ראשי במטריצה ריבועית

באלכסון הראשי מתקיים i == j. אין צורך בלולאה פנימית, אך אפשר לשלב בדיקה בתוך מעבר מלא.

static int MainDiagonalSum(int[][] mat)
{

    int n = mat.length;

    for (int i = 0; i < n; i++)

        sum += mat[i][i];
    }
    return sum;

תנאי: מתאים למטריצה ריבועית בלבד.

אלכסון משני

באלכסון המשני מתקיים j = n - 1 - i.

static int SecondaryDiagonalSum(int[][] mat)
{
    int sum = 0;
    int n = mat.length;

    for (int i = 0; i < n; i++)
    {
        int j = n - 1 - i;
        sum += mat[i][j];
    }
    return sum;
}

static boolean IsSymmetric(int[][] mat)
{
    int n = mat.length;
    boolean ok = true;

    for (int i = 0; i < n; i++)
    {
        for (int j = i + 1; j < n; j++)
        {
            if (mat[i][j] != mat[j][i])
                ok = false;
        }
    }
    return ok;
}

6. תבניות מיוחדות במטריצות

מסגרת המטריצה בלבד

תא נמצא במסגרת אם הוא בשורה ראשונה, שורה אחרונה, עמודה ראשונה או עמודה אחרונה.

static int SumBorder(int[][] mat)
{
    int sum = 0;
    int rows = mat.length;
    int cols = mat[0].length;

    for (int i = 0; i < rows; i++)
    {
        for (int j = 0; j < cols; j++)
        {
            if (i == 0 || i == rows - 1 || j == 0 || j == cols - 1)
                sum += mat[i][j];
        }
    }
    return sum;
}
שימוש: גבולות לוח משחק או מסגרת של תמונה.

שחלוף מטריצה

במטריצה משוחלפת השורות והעמודות מתחלפות: result[j][i] = mat[i][j].

static int[][] Transpose(int[][] mat)
{
    int rows = mat.length;
    int cols = mat[0].length;
    int[][] result = new int[cols][rows];

    for (int i = 0; i < rows; i++)
    {
        for (int j = 0; j < cols; j++)
        {
            result[j][i] = mat[i][j];
        }
    }
    return result;
}
דוגמה: מטריצה 2x3 הופכת למטריצה 3x2.

השוואת שתי מטריצות באותו גודל

עוברים על אותם מיקומים בשתי המטריצות. אם יש הבדל אחד, המטריצות אינן שוות.

static boolean EqualMatrices(int[][] a, int[][] b)
{
    if (a.length != b.length || a[0].length != b[0].length)
        return false;

    boolean equal = true;

    for (int i = 0; i < a.length; i++)
    {
        for (int j = 0; j < a[0].length; j++)
        {
            if (a[i][j] != b[i][j])
                equal = false;
        }
    }
    return equal;
}

כפל מטריצות — שלוש לולאות

לכל תא בתוצאה מחשבים סכום של מכפלות. זו תבנית קלאסית של לולאה מקוננת בתוך לולאה מקוננת.

static int[][] MultiplyMatrices(int[][] a, int[][] b)
{
    int rowsA = a.length;
    int colsA = a[0].length;
    int colsB = b[0].length;
    int[][] result = new int[rowsA][colsB];

    for (int i = 0; i < rowsA; i++)
    {
        for (int j = 0; j < colsB; j++)
        {
            int sum = 0;
            for (int k = 0; k < colsA; k++)
            {
                sum += a[i][k] * b[k][j];
            }
            result[i][j] = sum;
        }
    }
    return result;
}
תנאי: מספר העמודות של a חייב להיות שווה למספר השורות של b.

7. תבניות שכנים סביב תא במטריצה

בדיקת גבולות לפני גישה לתא

לפני שניגשים ל־mat[r][c], בודקים שהשורה והעמודה חוקיות.

static boolean IsInside(int[][] mat, int r, int c)
{
    return r >= 0 && r < mat.length &&
           c >= 0 && c < mat[0].length;
}
חשוב: תבנית זו מונעת חריגה מגבולות המערך.

ספירת שכנים חיוביים סביב תא

שתי לולאות קטנות עוברות על שינוי שורה ושינוי עמודה: -1, 0, 1.

static int CountPositiveNeighbors(int[][] mat, int row, int col)
{
    int count = 0;

    for (int dr = -1; dr <= 1; dr++)
    {
        for (int dc = -1; dc <= 1; dc++)
        {
            int r = row + dr;
            int c = col + dc;

            if (!(dr == 0 && dc == 0) && IsInside(mat, r, c))
            {
                if (mat[r][c] > 0)
                    count++;
            }
        }
    }
    return count;
}
שימושים: לוח מוקשים, משחק החיים, בדיקת סביבת תא.

החיצוניות עוברות על כל תא, והפנימיות מחשבות סביבו שכנים. זו תבנית חזקה לעיבוד גרידים.

static boolean HasDuplicates(int[] arr)
{
    boolean found = false;

    for (int i = 0; i < arr.length; i++)
    {
        for (int j = i + 1; j < arr.length; j++)
        {
            if (arr[i] == arr[j])
                found = true;
        }
    }
    return found;
}

static int MostFrequent(int[] arr)
{
    int bestValue = arr[0];
    int bestCount = 0;

    for (int i = 0; i < arr.length; i++)
    {
        int count = 0;
        for (int j = 0; j < arr.length; j++)
        {
            if (arr[j] == arr[i])
                count++;
        }

        if (count > bestCount)
        {
            bestCount = count;
            bestValue = arr[i];
        }
    }
    return bestValue;
}

9. מיון בעזרת לולאות מקוננות

מיון בחירה — Selection Sort

בכל סיבוב מחפשים את המקום של הערך הקטן ביותר בחלק שלא מוין, ומחליפים.

static void SelectionSort(int[] arr)
{
    for (int i = 0; i < arr.length - 1; i++)
    {
        int minIndex = i;

        for (int j = i + 1; j < arr.length; j++)
        {
            if (arr[j] < arr[minIndex])
                minIndex = j;
        }

        int temp = arr[i];
        arr[i] = arr[minIndex];
        arr[minIndex] = temp;
    }
}
הלולאה הפנימית תלויה בחיצונית: j = i + 1.

מיון בועות — Bubble Sort

בכל מעבר משווים איברים סמוכים. הערך הגדול מתקדם לקצה המערך.

static void BubbleSort(int[] arr)
{
    for (int pass = 0; pass < arr.length - 1; pass++)
    {
        for (int i = 0; i < arr.length - 1 - pass; i++)
        {
            if (arr[i] > arr[i + 1])
            {
                int temp = arr[i];
                arr[i] = arr[i + 1];
                arr[i + 1] = temp;
            }
        }
    }
}
תלות: בכל מעבר פנימי יש פחות השוואות, כי הקצה כבר מסודר.

static int MaxSubArraySumSimple(int[] arr)
{

    for (int start = 0; start < arr.length; start++)
    {

static boolean HasEqualDistancePair(int[] arr, int distance)
{
    boolean found = false;

    for (int i = 0; i < arr.length; i++)
    {
        for (int j = i + 1; j < arr.length; j++)
        {
            if (j - i == distance && arr[i] == arr[j])
                found = true;
        }
    }
    return found;
}
