מחלקת שירות היא מחלקה שמרכזת פעולות שימושיות כדי שאפשר יהיה לקרוא להן ממחלקות אחרות. בדרך כלל הפעולות אינן תלויות בעצם מסוים, ולכן כותבים אותן כפעולות static וקוראים להן בצורה ClassName.methodName(...).
במצגת מופיעים שני סוגי שירות מרכזיים: Math היא מחלקת שירות סטטית מובנית ב־Java ואין צורך לייבא אותה; Random היא מחלקה שממנה יוצרים עצם, ולכן יש לכתוב import java.util.Random;. הכלל החשוב: rnd.nextInt(n) מחזיר ערך בתחום 0 ≤ x < n, כלומר הגבול העליון אינו נכלל.

public class NumberService
{
    private NumberService()

    }

    public static boolean isEven(int n)
    {

    public static int lastDigit(int n)

    }
}
שימוש: System.out.println(NumberService.isEven(24)); // true

המרות טיפוסים עם Math

פעולות כמו Math.sqrt ו־Math.pow מחזירות בדרך כלל double. אם רוצים לשמור ב־int, חייבים לבצע המרה, אך החלק השבור נחתך.

int y = (int)Math.sqrt(45);     // 6
long r = Math.round(3.75);      // 4

עיגול למספר ספרות אחרי הנקודה

ב־Java אין פעולה מובנית בצורה Math.round(x, digits). בונים פעולה קטנה שמגדילה את המספר, מעגלת, ואז מחלקת בחזרה.

public static double roundTo(double x, int digits)
public static double roundTo(double x, int digits)
{
    double scale = Math.pow(10, digits);
    return Math.round(x * scale) / scale;
}
דוגמה: System.out.println(roundTo(3.756, 2)); // 3.76

public static double heronArea(double a, double b, double c)
{
    double p = (a + b + c) / 2;

}

public static double medianToA(double a, double b, double c)

5. אלגוריתם מלא: קלט, חישוב, פלט

תוכנית לדוגמה — משולש ישר זווית

הפעולות נמצאות במחלקת שירות, ו־main אחראית על הקלט והפלט.

import java.util.Scanner;

class GeometryService
{
    public static double hypotenuse(double a, double b)
    {
        return Math.sqrt(Math.pow(a, 2) + Math.pow(b, 2));
    }

    public static double rightTriangleArea(double a, double b)
    {
        return (a * b) / 2;
    }

    public static double rightTrianglePerimeter(double a, double b)
    {
        double c = hypotenuse(a, b);
        return a + b + c;
    }
}

public class Main
{
    public static void main(String[] args)
    {
        Scanner input = new Scanner(System.in);

        System.out.print("Enter a: ");
        double a = input.nextDouble();

        System.out.print("Enter b: ");
        double b = input.nextDouble();

        double c = GeometryService.hypotenuse(a, b);
        double area = GeometryService.rightTriangleArea(a, b);
        double perimeter = GeometryService.rightTrianglePerimeter(a, b);

        System.out.println("Hypotenuse = " + c);
        System.out.println("Area = " + area);
        System.out.println("Perimeter = " + perimeter);
    }
}

public static int countHeads(Random rnd, int times)
{
    int count = 0;
    for (int i = 0; i < times; i++)

}

8. יצירת מחלקת שירות להגרלות

מחלקת שירות RandomService

המחלקה מקבלת את העצם Random כפרמטר. כך לא יוצרים אותו מחדש בכל פעולה.

import java.util.Random;

public class RandomService
{
    private RandomService()
    {
    }

    public static int randomInRange(Random rnd, int min, int max)
    {
        return rnd.nextInt(max - min + 1) + min;
    }

    public static char randomUpperLetter(Random rnd)
    {
        return (char)(rnd.nextInt('Z' - 'A' + 1) + 'A');
    }

    public static char randomLowerLetter(Random rnd)
    {
        return (char)(rnd.nextInt('z' - 'a' + 1) + 'a');
    }

    public static char randomDigit(Random rnd)
    {
        return (char)(rnd.nextInt('9' - '0' + 1) + '0');
    }

    public static char randomSpecial(Random rnd)
    {
        String symbols = "!@#$%^&*";
        int index = rnd.nextInt(symbols.length());
        return symbols.charAt(index);
    }
}

אלגוריתם סיסמה לפי מבנה קבוע

מבנה לדוגמה: 2 אותיות גדולות, 2 ספרות, 2 תווים מיוחדים, 6 אותיות קטנות.

public static String buildPassword(Random rnd)
{
    String password = "";

    for (int i = 0; i < 2; i++)
        password = password + RandomService.randomUpperLetter(rnd);

    for (int i = 0; i < 2; i++)
        password = password + RandomService.randomDigit(rnd);

    for (int i = 0; i < 2; i++)
        password = password + RandomService.randomSpecial(rnd);

    for (int i = 0; i < 6; i++)
        password = password + RandomService.randomLowerLetter(rnd);

    return password;
}
דוגמה: System.out.println(buildPassword(rnd));
