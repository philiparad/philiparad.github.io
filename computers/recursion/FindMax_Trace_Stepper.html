static int FindMax(int[] array, int left, int right)
{
    if (left == right)
    {
        return array[left];
    }
    int mid = (left + right) / 2;
    int maxLeft = FindMax(array, left, mid);
    int maxRight = FindMax(array, mid + 1, right);
    return Math.Max(maxLeft, maxRight);
}
