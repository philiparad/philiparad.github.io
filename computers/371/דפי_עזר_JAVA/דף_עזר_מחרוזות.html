static boolean areEqual(String a, String b)
{

4. בדיקת התאמה של תת־מחרוזת בלי substring

הרעיון: בודקים תו־תו האם value מתחילה במקום index בתוך str.

האם יש התאמה במקום מסוים?

static boolean isMatchAt(String str, String value, int index)
static boolean isMatchAt(String str, String value, int index)
{
    if (index < 0 || index + value.length() > str.length())
        return false;

    boolean ok = true;
    for (int j = 0; j < value.length(); j++)
    {
        if (str.charAt(index + j) != value.charAt(j))
            ok = false;
    }
    return ok;
}
דוגמה: System.out.println(isMatchAt("banana", "na", 2)); // true

מציאת תת־מחרוזת בצורה ידנית

static int indexOfText(String str, String value)
static int indexOfText(String str, String value)
{
    if (value.length() == 0)
        return 0;

    int index = -1;
    for (int i = 0; i <= str.length() - value.length() && index == -1; i++)
    {
        if (isMatchAt(str, value, i))
            index = i;
    }
    return index;
}
דוגמה: System.out.println(indexOfText("banana", "na")); // 2

static int countText(String str, String value)

static boolean endsWithText(String str, String suffix)
{
    if (suffix.length() > str.length())
        return false;

    int start = str.length() - suffix.length();
    boolean ok = true;
    for (int i = 0; i < suffix.length(); i++)

            ok = false;
    }
    return ok;
}
דוגמה: System.out.println(endsWithText("computer", "ter")); // true

יצירת חלק ממחרוזת בלי substring

הפעולה מחזירה מחרוזת באורך length, שמתחילה במקום start. אם הנתונים לא חוקיים — מוחזרת מחרוזת ריקה.

static String mySubstring(String str, int start, int length)
static String mySubstring(String str, int start, int length)
{
    String result = "";

    if (start < 0 || length < 0 || start + length > str.length())
        return result;

    for (int i = start; i < start + length; i++)
    {
        result = result + str.charAt(i);
    }
    return result;
}
דוגמה: System.out.println(mySubstring("computer", 3, 4)); // pute

static String replaceChar(String str, char oldCh, char newCh)
{
    String result = "";
    for (int i = 0; i < str.length(); i++)
    {
        if (str.charAt(i) == oldCh)

}

static String deleteAt(String str, int index)
static String deleteAt(String str, int index)
{
    if (index < 0 || index >= str.length())
        return str;

    String result = "";
    for (int i = 0; i < str.length(); i++)
    {
        if (i != index)
            result = result + str.charAt(i);
    }
    return result;
}
דוגמה: System.out.println(deleteAt("hello", 1)); // hllo

הוספת מחרוזת במקום מסוים

static String insertAt(String str, String value, int index)
static String insertAt(String str, String value, int index)
{
    if (index < 0 || index > str.length())
        return str;

    String result = "";
    for (int i = 0; i < index; i++)
        result = result + str.charAt(i);

    result = result + value;

    for (int i = index; i < str.length(); i++)
        result = result + str.charAt(i);

    return result;
}
דוגמה: System.out.println(insertAt("helo", "l", 3)); // hello

7. פעולות בדיקה נפוצות

בדיקת פלינדרום

פלינדרום הוא מחרוזת שנקראת אותו הדבר משני הכיוונים.

static boolean isPalindrome(String str)
static boolean isPalindrome(String str)
{
    boolean ok = true;
    int left = 0;
    int right = str.length() - 1;

    while (left < right)
    {
        if (str.charAt(left) != str.charAt(right))
            ok = false;

        left++;
        right--;
    }
    return ok;
}
דוגמה: System.out.println(isPalindrome("level")); // true

ספירת מילים לפי רווחים

מתאים למילים שמופרדות ברווח אחד או יותר. אין צורך ב־trim.

static int countWords(String str)
static int countWords(String str)
{
    int count = 0;
    boolean insideWord = false;

    for (int i = 0; i < str.length(); i++)
    {
        if (str.charAt(i) != ' ' && !insideWord)
        {
            count++;
            insideWord = true;
        }
        else if (str.charAt(i) == ' ')
        {
            insideWord = false;
        }
    }
    return count;
}
דוגמה: System.out.println(countWords("one two three")); // 3

השוואה מילונית בלי compareTo

-1 אם a לפני b, 1 אם אחרי, 0 אם שוות.

static int compareStrings(String a, String b)
static int compareStrings(String a, String b)
{
    int i = 0;
    int result = 0;

    while (i < a.length() && i < b.length() && result == 0)
    {
        if (a.charAt(i) < b.charAt(i))
            result = -1;
        else if (a.charAt(i) > b.charAt(i))
            result = 1;

        i++;
    }

    if (result == 0)
    {
        if (a.length() < b.length())
            result = -1;
        else if (a.length() > b.length())
            result = 1;
    }
    return result;
}
דוגמה: System.out.println(compareStrings("abc", "abd")); // -1

המרת אות גדולה לקטנה בלי toLowerCase

הפעולה מטפלת באותיות באנגלית בלבד. כל תו אחר נשאר כפי שהוא.

static char toLowerChar(char ch)
static char toLowerChar(char ch)
{
    if (ch >= 'A' && ch <= 'Z')
        return (char)(ch + ('a' - 'A'));

    return ch;
}
דוגמה: System.out.println(toLowerChar('G')); // g
static String toLowerText(String str)
static String toLowerText(String str)
{
    String result = "";
    for (int i = 0; i < str.length(); i++)
        result = result + toLowerChar(str.charAt(i));
    return result;
}
דוגמה: System.out.println(toLowerText("AbC!")); // abc!
