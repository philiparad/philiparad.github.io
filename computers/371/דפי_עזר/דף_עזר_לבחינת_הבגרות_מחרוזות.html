4. בדיקת התאמה של תת־מחרוזת בלי Substring

הרעיון: בודקים תו־תו האם value מתחילה במקום index בתוך str.

האם יש התאמה במקום מסוים?

static bool IsMatchAt(string str, string value, int index)
static bool IsMatchAt(string str, string value, int index)
{
    if (index < 0 || index + value.Length > str.Length)
        return false;

    bool ok = true;
    for (int j = 0; j < value.Length; j++)
    {
        if (str[index + j] != value[j])
            ok = false;
    }
    return ok;
}
דוגמה: Console.WriteLine(IsMatchAt("banana", "na", 2)); // True

מציאת תת־מחרוזת בצורה ידנית

static int IndexOfText(string str, string value)
static int IndexOfText(string str, string value)
{
    if (value.Length == 0)
        return 0;

    int index = -1;
    for (int i = 0; i <= str.Length - value.Length && index == -1; i++)
    {
        if (IsMatchAt(str, value, i))
            index = i;
    }
    return index;
}
דוגמה: Console.WriteLine(IndexOfText("banana", "na")); // 2

ספירת תת־מחרוזת

הקוד סופר גם הופעות חופפות. לדוגמה: ב־"aaaa", המחרוזת "aa" נספרת 3 פעמים.

static int CountText(string str, string value)
{
    if (value.Length == 0)
        return 0;

    int count = 0;
    for (int i = 0; i <= str.Length - value.Length; i++)

}

יצירת חלק ממחרוזת בלי Substring

הפעולה מחזירה מחרוזת באורך length, שמתחילה במקום start. אם הנתונים לא חוקיים — מוחזרת מחרוזת ריקה.

static string MySubstring(string str, int start, int length)
static string MySubstring(string str, int start, int length)
{
    string result = "";

    if (start < 0 || length < 0 || start + length > str.Length)
        return result;

    for (int i = start; i < start + length; i++)
    {
        result = result + str[i];
    }
    return result;
}
דוגמה: Console.WriteLine(MySubstring("computer", 3, 4)); // pute

static string Reverse(string str)
{
    string result = "";
    for (int i = str.Length - 1; i >= 0; i--)

        result = result + str[i];
    }
    return result;
}

static string ReplaceChar(string str, char oldCh, char newCh)
{
    string result = "";
    for (int i = 0; i < str.Length; i++)
    {
        if (str[i] == oldCh)
            result = result + newCh;
        else
            result = result + str[i];
    }
    return result;
}
דוגמה: Console.WriteLine(ReplaceChar("banana", 'a', 'o')); // bonono

מחיקת תו במקום מסוים

static string DeleteAt(string str, int index)
{
    if (index < 0 || index >= str.Length)
        return str;

    string result = "";
    for (int i = 0; i < str.Length; i++)
    {
        if (i != index)
            result = result + str[i];
    }
    return result;
}
דוגמה: Console.WriteLine(DeleteAt("hello", 1)); // hllo

הוספת מחרוזת במקום מסוים

static string InsertAt(string str, string value, int index)
{
    if (index < 0 || index > str.Length)
        return str;

    string result = "";
    for (int i = 0; i < index; i++)
        result = result + str[i];

    result = result + value;

    for (int i = index; i < str.Length; i++)
        result = result + str[i];

    return result;
}
דוגמה: Console.WriteLine(InsertAt("helo", "l", 3)); // hello

7. פעולות בדיקה נפוצות

בדיקת פלינדרום

פלינדרום הוא מחרוזת שנקראת אותו הדבר משני הכיוונים.

static bool IsPalindrome(string str)
static bool IsPalindrome(string str)
{
    bool ok = true;
    int left = 0;
    int right = str.Length - 1;

    while (left < right)
    {
        if (str[left] != str[right])
            ok = false;

        left++;
        right--;
    }
    return ok;
}
דוגמה: Console.WriteLine(IsPalindrome("level")); // True

ספירת מילים לפי רווחים

מתאים למילים שמופרדות ברווח אחד או יותר. אין צורך ב־Trim.

static int CountWords(string str)
static int CountWords(string str)
{
    int count = 0;
    bool insideWord = false;

    for (int i = 0; i < str.Length; i++)
    {
        if (str[i] != ' ' && !insideWord)
        {
            count++;
            insideWord = true;
        }
        else if (str[i] == ' ')
        {
            insideWord = false;
        }
    }
    return count;
}
דוגמה: Console.WriteLine(CountWords("one two three")); // 3

השוואה מילונית בלי CompareTo

-1 אם a לפני b, 1 אם אחרי, 0 אם שוות.

static int CompareStrings(string a, string b)
static int CompareStrings(string a, string b)
{
    int i = 0;
    int result = 0;

    while (i < a.Length && i < b.Length && result == 0)
    {
        if (a[i] < b[i])
            result = -1;
        else if (a[i] > b[i])
            result = 1;

        i++;
    }

    if (result == 0)
    {
        if (a.Length < b.Length)
            result = -1;
        else if (a.Length > b.Length)
            result = 1;
    }
    return result;
}
דוגמה: Console.WriteLine(CompareStrings("abc", "abd")); // -1

המרת אות גדולה לקטנה בלי ToLower

הפעולה מטפלת באותיות באנגלית בלבד. כל תו אחר נשאר כפי שהוא.

static char ToLowerChar(char ch)
static char ToLowerChar(char ch)
{
    if (ch >= 'A' && ch <= 'Z')
        return (char)(ch + ('a' - 'A'));

    return ch;
}
דוגמה: Console.WriteLine(ToLowerChar('G')); // g
static string ToLowerText(string str)
static string ToLowerText(string str)
{
    string result = "";
    for (int i = 0; i < str.Length; i++)
        result = result + ToLowerChar(str[i]);
    return result;
}
דוגמה: Console.WriteLine(ToLowerText("AbC!")); // abc!
